1. Why "No Scholarships Found" happens
The search code I could see shows four likely causes. Run flask scholarships diagnose (added below) to see which one applies to your database:

No record has been both verified and published. The CSV importer creates every row as unverified and unpublished on purpose. Students only see published records, so after an import nobody sees anything until an admin verifies and publishes. This is the most likely cause.
Expired records are filtered out. Any record whose closing date has passed is hidden by default.
Education level filtering was case-sensitive. A profile value like Undergraduate didn't match a criterion stored as undergraduate. The new search compares in lower case.
The frontend may be reading the response wrongly. If the API returns {data: {items}} and the page reads res.data.items, it gets an empty list. The new asPage() helper accepts all common response shapes.
The app does not add any placeholder scholarships. If the diagnosis shows zero published records, the empty message is correct, and the fix is for an admin to enter real records and verify them.

from __future__ import annotations

from dataclasses import dataclass
from decimal import ROUND_CEILING, ROUND_DOWN, ROUND_HALF_UP, Decimal, InvalidOperation

ROUNDING = {"half_up": ROUND_HALF_UP, "truncate": ROUND_DOWN}
MARKS_ROUNDING = {"none": None, "ceil": ROUND_CEILING, "half_up": ROUND_HALF_UP}


class CalcError(ValueError):
    def __init__(self, errors: dict):
        super().__init__("Please correct the highlighted fields")
        self.errors = errors


@dataclass(frozen=True)
class Band:
    label: str
    grade_point: Decimal
    is_pass: bool
    min_marks: Decimal | None = None
    max_marks: Decimal | None = None
    is_absent: bool = False


@dataclass(frozen=True)
class SchemeRules:
    bands: tuple
    max_total_marks: Decimal = Decimal("100")
    max_grade_point: Decimal = Decimal("10")
    cie_max: Decimal | None = None
    see_max: Decimal | None = None
    cie_min_pass: Decimal | None = None
    see_min_pass: Decimal | None = None
    total_min_pass: Decimal | None = None
    decimal_places: int = 2
    rounding: str = "half_up"
    marks_rounding: str = "none"
    fail_credits_in_denominator: bool = True
    max_subject_credits: Decimal = Decimal("20")


def s(d: Decimal | None) -> str | None:
    return None if d is None else format(d.normalize(), "f")


def round_to(value: Decimal, places: int, mode: str) -> Decimal:
    return value.quantize(Decimal(1).scaleb(-places), rounding=ROUNDING.get(mode, ROUND_HALF_UP))


def num(value, path, errors, *, lo=None, hi=None, required=True):
    if value is None or (isinstance(value, str) and not value.strip()):
        if required:
            errors[path] = "Required"
        return None
    if isinstance(value, bool):
        errors[path] = "Must be a number"
        return None
    try:
        d = Decimal(str(value).strip())
    except (InvalidOperation, ValueError):
        errors[path] = "Must be a number"
        return None
    if not d.is_finite():
        errors[path] = "Must be a number"
        return None
    if lo is not None and d < Decimal(str(lo)):
        errors[path] = f"Must be at least {lo}"
        return None
    if hi is not None and d > Decimal(str(hi)):
        errors[path] = f"Must be at most {s(Decimal(str(hi)))}"
        return None
    return d


def _fail_band(rules):
    fails = [b for b in rules.bands if not b.is_pass and not b.is_absent]
    if not fails:
        raise CalcError({"scheme": "The scheme has no failing grade configured"})
    return min(fails, key=lambda b: b.grade_point)


def _absent_band(rules):
    return next((b for b in rules.bands if b.is_absent), None) or _fail_band(rules)


def band_for_marks(rules, marks):
    for b in rules.bands:
        if not b.is_absent and b.min_marks is not None and b.max_marks is not None and b.min_marks <= marks <= b.max_marks:
            return b
    return None


def _grade_from_marks(rules, subj, p, errors):
    split = subj.get("cie") not in (None, "") or subj.get("see") not in (None, "")
    cie = see = None
    if split:
        if rules.cie_max is None or rules.see_max is None:
            errors[f"{p}.cie"] = "This scheme has no CIE/SEE split configured. Enter total marks instead."
            return None, None, None
        cie = num(subj.get("cie"), f"{p}.cie", errors, lo=0, hi=rules.cie_max)
        see = num(subj.get("see"), f"{p}.see", errors, lo=0, hi=rules.see_max)
        if cie is None or see is None:
            return None, None, None
        total = cie + see
    else:
        total = num(subj.get("total"), f"{p}.total", errors, lo=0, hi=rules.max_total_marks)
        if total is None:
            return None, None, None
    mode = MARKS_ROUNDING.get(rules.marks_rounding)
    graded = total.quantize(Decimal(1), rounding=mode) if mode else total
    reasons = []
    if cie is not None and rules.cie_min_pass is not None and cie < rules.cie_min_pass:
        reasons.append(f"CIE {s(cie)} is below the minimum {s(rules.cie_min_pass)}")
    if see is not None and rules.see_min_pass is not None and see < rules.see_min_pass:
        reasons.append(f"SEE {s(see)} is below the minimum {s(rules.see_min_pass)}")
    if rules.total_min_pass is not None and graded < rules.total_min_pass:
        reasons.append(f"Total {s(graded)} is below the minimum {s(rules.total_min_pass)}")
    if reasons:
        return _fail_band(rules), total, "; ".join(reasons)
    band = band_for_marks(rules, graded)
    if band is None:
        errors[f"{p}.total"] = ("These marks fall between grade bands and this scheme has no marks-rounding rule "
                                "configured. Select the grade from your marks card instead."
                                if graded != graded.to_integral_value() else "No grade band covers these marks")
        return None, None, None
    return band, total, None


def compute_sgpa(rules: SchemeRules, subjects) -> dict:
    if not isinstance(subjects, list) or not subjects:
        raise CalcError({"subjects": "Add at least one subject"})
    if len(subjects) > 40:
        raise CalcError({"subjects": "At most 40 subjects per semester"})
    labels = {b.label.upper(): b for b in rules.bands}
    errors, rows, codes = {}, [], set()
    for i, subj in enumerate(subjects):
        p = f"subjects[{i}]"
        if not isinstance(subj, dict):
            errors[p] = "Invalid subject"
            continue
        name = str(subj.get("name") or "").strip()[:160]
        code = str(subj.get("code") or "").strip().upper()[:20]
        if not name and not code:
            errors[f"{p}.name"] = "Enter a subject name or code"
        if code:
            if code in codes:
                errors[f"{p}.code"] = "This subject code is entered twice"
            codes.add(code)
        credits = num(subj.get("credits"), f"{p}.credits", errors, lo=0, hi=rules.max_subject_credits)
        if credits is not None and (credits * 2) % 1 != 0:
            errors[f"{p}.credits"] = "Credits must be a whole or half number"
            credits = None
        band, total, note = None, None, None
        mode = subj.get("mode") or "grade"
        if subj.get("absent"):
            band, note = _absent_band(rules), "Absent"
        elif mode == "grade":
            g = str(subj.get("grade") or "").strip().upper()
            if not g:
                errors[f"{p}.grade"] = "Select a grade"
            elif g not in labels:
                errors[f"{p}.grade"] = f"'{g}' is not a grade in this scheme"
            else:
                band = labels[g]
        elif mode == "marks":
            band, total, note = _grade_from_marks(rules, subj, p, errors)
        else:
            errors[f"{p}.mode"] = "Choose grade or marks"
        if band is None or credits is None:
            continue
        counted = credits > 0 and (band.is_pass or rules.fail_credits_in_denominator)
        rows.append({"name": name, "code": code, "credits": credits, "grade": band.label,
                     "grade_point": band.grade_point, "credit_points": credits * band.grade_point,
                     "passed": band.is_pass, "absent": bool(subj.get("absent")), "total_marks": total,
                     "note": note, "counted": counted})
    if errors:
        raise CalcError(errors)
    denom = sum((r["credits"] for r in rows if r["counted"]), Decimal(0))
    points = sum((r["credit_points"] for r in rows if r["counted"]), Decimal(0))
    if denom == 0:
        raise CalcError({"subjects": "Total credits must be greater than zero"})
    raw = points / denom
    failed = [r for r in rows if not r["passed"]]
    assumptions = []
    if any(r["credits"] == 0 for r in rows):
        assumptions.append("Zero-credit subjects are listed but not included in the SGPA.")
    if failed:
        assumptions.append("Failed or absent subjects carry 0 grade points; their credits are "
                           + ("included in" if rules.fail_credits_in_denominator else "excluded from")
                           + " total credits, as configured for this scheme.")
    return {
        "rows": [{**r, "credits": s(r["credits"]), "grade_point": s(r["grade_point"]),
                  "credit_points": s(r["credit_points"]), "total_marks": s(r["total_marks"])} for r in rows],
        "registered_credits": s(sum((r["credits"] for r in rows), Decimal(0))),
        "total_credits": s(denom), "total_credit_points": s(points),
        "sgpa_exact": str(raw), "sgpa": str(round_to(raw, rules.decimal_places, rules.rounding)),
        "all_passed": not failed,
        "result_status": "Passed all subjects" if not failed else f"Not cleared: {len(failed)} subject(s)",
        "assumptions": assumptions,
    }


def compute_manual_sgpa(subjects, max_grade_point=Decimal("10")) -> dict:

        raise CalcError({"subjects": "Add at least one subject"})
    errors, rows = {}, []
    for i, subj in enumerate(subjects[:40]):

        gp = num((subj or {}).get("grade_point"), f"{p}.grade_point", errors, lo=0, hi=max_grade_point)
        if c is not None and gp is not None:

    if denom == 0:
        raise CalcError({"subjects": "Total credits must be greater than zero"})
    points = sum((r["credit_points"] for r in rows if r["counted"]), Decimal(0))
    raw = points / denom
    return {"rows": [{**r, "credits": s(r["credits"]), "grade_point": s(r["grade_point"]),
                      "credit_points": s(r["credit_points"])} for r in rows],
            "registered_credits": s(denom), "total_credits": s(denom), "total_credit_points": s(points),
            "sgpa_exact": str(raw), "sgpa": str(round_to(raw, 2, "half_up")), "all_passed": None,
            "result_status": "Not determined (manual grade points)",
            "assumptions": ["Manual mode: SGPA = Σ(credits × grade point) ÷ Σ credits using the grade points you "
                            "entered. Pass/fail and scheme rules were not applied; rounded half-up to 2 decimals."]}


def compute_cgpa(semesters, *, max_grade_point=Decimal("10"), decimal_places=2, rounding="half_up",
                 percentage_formula=None, classification=None) -> dict:
    if not isinstance(semesters, list) or not semesters:
        raise CalcError({"semesters": "Add at least one semester"})
    errors, rows, seen = {}, [], set()
    for i, sem in enumerate(semesters[:12]):
        p = f"semesters[{i}]"
        sem = sem if isinstance(sem, dict) else {}
        n = num(sem.get("semester"), f"{p}.semester", errors, lo=1, hi=12)
        if n is not None:
            if n != n.to_integral_value():
                errors[f"{p}.semester"] = "Semester must be a whole number"
                n = None
            elif int(n) in seen:
                errors[f"{p}.semester"] = "Semester entered twice"
            else:
                seen.add(int(n))
        sg = num(sem.get("sgpa"), f"{p}.sgpa", errors, lo=0, hi=max_grade_point)
        cr = num(sem.get("credits"), f"{p}.credits", errors, lo=Decimal("0.5"), hi=60)
        if None not in (n, sg, cr):
            rows.append({"semester": int(n), "sgpa": sg, "credits": cr, "credit_points": sg * cr})
    if errors:
        raise CalcError(errors)
    rows.sort(key=lambda r: r["semester"])
    tc = sum((r["credits"] for r in rows), Decimal(0))
    tp = sum((r["credit_points"] for r in rows), Decimal(0))
    raw = tp / tc
    cgpa = round_to(raw, decimal_places, rounding)
    assumptions = ["CGPA = Σ(semester SGPA × semester credits) ÷ Σ semester credits (credit-weighted, not a simple average).",
                   "SGPAs you enter are usually already rounded, so the result can differ slightly from a CGPA "
                   "computed from subject-level credit points."]
    missing = sorted(set(range(1, max(seen) + 1)) - seen)
    if missing:
        assumptions.append(f"Semester(s) {', '.join(map(str, missing))} not entered; CGPA covers entered semesters only.")
    result = {"rows": [{k: (s(v) if isinstance(v, Decimal) else v) for k, v in r.items()} for r in rows],
              "total_credits": s(tc), "total_credit_points": s(tp), "cgpa_exact": str(raw), "cgpa": str(cgpa),
              "percentage": None, "percentage_basis": None, "classification": None, "assumptions": assumptions}
    if percentage_formula and percentage_formula.get("type") == "linear" and percentage_formula.get("source_url"):
        pct = (cgpa - Decimal(str(percentage_formula.get("subtract", 0)))) * Decimal(str(percentage_formula.get("multiply", 1)))
        result["percentage"] = str(round_to(min(max(pct, Decimal(0)), Decimal(100)), 2, "half_up"))
        result["percentage_basis"] = {"formula": f"(CGPA − {percentage_formula.get('subtract', 0)}) × {percentage_formula.get('multiply', 1)}",
                                      "source_url": percentage_formula["source_url"], "applied_to": "rounded CGPA"}
    for c in classification or []:
        lo = Decimal(str(c.get("min", 0)))
        hi = Decimal(str(c["max"])) if c.get("max") is not None else None
        if cgpa >= lo and (hi is None or cgpa <= hi):
            result["classification"] = c.get("label")
            break
    return result

def _clean(kind, payload, partial=False):
    Model, spec, required = _kind(kind)
    out, errors = {}, {}
    for f, typ in spec.items():
        if f not in payload:
            continue
        v = payload[f]
        if v in ("", None):
            out[f] = None
        elif typ is int:
            try:
                out[f] = int(v)
            except (TypeError, ValueError):
                errors[f] = ["Must be a number"]
        elif typ is bool:
            out[f] = bool(v)
        else:
            v = str(v).strip()
            length = Model.__table__.columns[f].type.length
            if length and len(v) > length:
                errors[f] = [f"At most {length} characters"]
            out[f] = v
    if not partial:
        for f in required:
            if out.get(f) in (None, ""):
                errors[f] = ["Required"]
    if out.get("state") and out["state"] not in INDIAN_STATES_AND_UTS:
        errors["state"] = ["Unknown state/UT"]
    if out.get("source_url") and not URL.match(out["source_url"]):
        errors["source_url"] = ["Must be an http(s) URL"]
    for f, Parent in PARENTS.items():
        if out.get(f) and not db.session.get(Parent, out[f]):
            errors[f] = ["Not found"]
    if errors:
        raise ValidationError(errors)
    return out

Copyexport type Page<T> = { items: T[]; total: number; page: number; pages: number };

export function unwrap<T = any>(res: any): T {
  const body = res?.data ?? res;
  return (body && typeof body === "object" && "data" in body ? body.data : body) as T;
}

export function asPage<T>(res: any): Page<T> {
  const d: any = unwrap(res);
  const items: T[] = Array.isArray(d) ? d : d?.items ?? d?.results ?? d?.scholarships ?? [];
  const meta = d?.meta ?? res?.data?.meta ?? {};
  const total = meta.total ?? d?.total ?? items.length;
  const per = meta.per_page ?? d?.per_page ?? (items.length || 1);
  return { items, total, page: meta.page ?? d?.page ?? 1, pages: meta.pages ?? Math.max(1, Math.ceil(total / per)) };
}

export function fieldErrors(err: any): Record<string, string> {
  const e = err?.response?.data?.error ?? err?.response?.data ?? {};
  const f = e.fields ?? e.errors ?? e.details ?? {};
  return Object.fromEntries(Object.entries(f).map(([k, v]) => [k, Array.isArray(v) ? String(v[0]) : String(v)]));
}

export function errorMessage(err: any, fallback = "Something went wrong"): string {
  return err?.response?.data?.error?.message ?? err?.response?.data?.message ?? err?.message ?? fallback;
}

import api from "../api/client"; // CHECK: your axios instance path
import { unwrap } from "../lib/api-helpers";
import type { Opt } from "../components/SearchSelect";

const cache = new Map<string, any[]>();

export function useReference(kind: string, params: Record<string, any> = {}, enabled = true) {
  const key = `${kind}?${new URLSearchParams(Object.entries(params).filter(([, v]) => v != null && v !== "") as any)}`;
  const [items, setItems] = useState<any[]>(cache.get(key) ?? []);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  useEffect(() => {
    if (!enabled) { setItems([]); return; }
    if (cache.has(key)) { setItems(cache.get(key)!); return; }
    let alive = true;
    setLoading(true); setError(null);
    api.get(`/reference/${key}`)
      .then((r) => { const d = unwrap<any[]>(r) ?? []; cache.set(key, d); alive && setItems(d); })
      .catch(() => alive && setError("Could not load options"))
      .finally(() => alive && setLoading(false));
    return () => { alive = false; };
  }, [key, enabled]);
  const options: Opt[] = items.map((i) => ({ value: i.id, label: i.name, hint: i.short_name ?? i.college_code ?? undefined }));
  return { items, options, loading, error };
}

export function useRefMeta() {
  const [meta, setMeta] = useState<any>(cache.get("__meta")?.[0] ?? null);
  useEffect(() => {
    if (meta) return;
    api.get("/reference/meta").then((r) => { const d = unwrap(r); cache.set("__meta", [d]); setMeta(d); }).catch(() => {});
  }, [meta]);
  return meta;
}
